1. Executive Summary & Design Philosophy
Dex Sprite is a professional, offline macOS desktop utility engineered to convert modern high-resolution animation sequences into retro-accurate, hardware-constrained Super Nintendo (SNES) sprite sheets, diagnostic looping GIFs, and instant HTML5 playtest code.
Core Problems Solved:
* The Consistency Problem: Generative AI engines cannot maintain frame-to-frame pixel registration. Dex Sprite uses deterministic mathematics to guarantee pixel-perfect stability across animation cycles.
* The Drift Problem (Manual Registration): If input frames are of varying sizes or slightly off-center, standard compilers produce "jittery" animations. Dex Sprite introduces an interactive crosshair registration tool to manually offset and anchor each frame's pivot point.
* The Gradient Problem (Retro Dithering): Downsampling smooth modern gradients to a strict 15-color palette produces ugly color banding. Dex Sprite implements classic Bayer    and Floyd-Steinberg dithering algorithms to recreate authentic retro shading.

   def load_directory(self):

       if not selected:
           return
           
       self.source_dir = selected

       # Scan for standardized image extensions
       extensions = ('.png', '.jpg', '.jpeg', '.bmp', '.tiff', '.webp')
       self.frames_list = sorted([
           os.path.join(self.source_dir, f) 
           for f in os.listdir(self.source_dir) 
           if f.lower().endswith(extensions)

           messagebox.showerror("No Images Found", "The selected folder does not contain any compatible image formats.")
           return
           
       # Initialize Frame Timeline variables
       self.active_frames = [True] * len(self.frames_list)
       self.offsets = [(0, 0)] * len(self.frames_list)
       self.selected_timeline_index = 0
       
       self.status_var.set(f"Loaded {len(self.frames_list)} frames. Click on preview panel to manually drag/align pixel drift.")
       self.recompute()
       
   def recompute(self):
       """Main pipeline: resizes, quantizes, applies dithering and compiles the spritesheet."""
       if not self.frames_list:
           return
           
       if self.animation_job:
           self.root.after_cancel(self.animation_job)
           self.animation_job = None
           
       # Parse configurations
       size_str = self.size_var.get()
       w_str, h_str = size_str.split("x")
       sprite_w, sprite_h = int(w_str), int(h_str)
       padding = self.pad_var.get()
       dither_type = self.dither_var.get()
       
       # 1. Sequential Image Downsampling (Nearest-Neighbor)
       raw_resized_frames = []
       for i, file_path in enumerate(self.frames_list):
           if self.active_frames[i]:
               try:
                   img = Image.open(file_path).convert("RGB")
                   
                   # Cut out crop box using manually set timeline offsets (dx, dy)
                   dx, dy = self.offsets[i]
                   crop_box = (dx, dy, dx + img.width, dy + img.height)
                   offset_canvas = Image.new("RGB", img.size, color=(128, 128, 128))
                   offset_canvas.paste(img.crop(crop_box), (0,0))
                   
                   # Convert via Nearest Neighbor
                   resized_img = offset_canvas.resize((sprite_w, sprite_h), Image.Resampling.NEAREST)
                   raw_resized_frames.append(resized_img)
               except Exception as e:
                   print(f"Failed to process frame {file_path}: {e}")
                   
       if not raw_resized_frames:
           self.status_var.set("Error: All frames inactive. Please check at least one timeline frame.")
           return
           
       # 2. Build Multi-frame composite in-memory to resolve global palette variables
       combined_width = sprite_w * len(raw_resized_frames)
       comp_canvas = Image.new("RGB", (combined_width, sprite_h))
       for idx, img in enumerate(raw_resized_frames):
           comp_canvas.paste(img, (idx * sprite_w, 0))
           
       # 3. Median Cut Clustering down to 15 colors
       global_quantized = comp_canvas.quantize(colors=15, method=Image.Quantize.MEDIANCUT)
       raw_palette_bytes = global_quantized.getpalette()[:45]
       
       # 4. Clamp raw 24-bit RGB values to 15-bit retro space
       clamped_palette = []
       for val in raw_palette_bytes:
           five_bit_val = val // 8
           clamped_val = five_bit_val * 8
           clamped_palette.append(clamped_val)
           
       # Set transparent color index (15) to classic Magenta #FF00FF
       clamped_palette.extend([255, 0, 255])
       while len(clamped_palette) < 768:
           clamped_palette.extend([0, 0, 0])
           
       self.master_palette = clamped_palette
       
       # Build strict palette template structure
       pal_template = Image.new("P", (1, 1))
       pal_template.putpalette(self.master_palette)
       
       # 5. Apply quantization and configured dither matrix
       self.processed_frames = []
       for raw_frame in raw_resized_frames:
           if dither_type == "none":
               clamped_frame = raw_frame.quantize(palette=pal_template, dither=Image.Dither.NONE)
           elif dither_type == "floyd":
               clamped_frame = raw_frame.quantize(palette=pal_template, dither=Image.Dither.FLOYDSTEINBERG)
           elif dither_type == "bayer":
               # Manual implementation of Bayer 4x4 Ordered Matrix Dithering
               clamped_frame = self.apply_bayer_dither(raw_frame, pal_template)
           else:
               clamped_frame = raw_frame.quantize(palette=pal_template, dither=Image.Dither.NONE)
               
           self.processed_frames.append(clamped_frame)
           
       # 6. Assemble Sprite Sheet Output Canvas Map
       num_frames = len(self.processed_frames)
       cols = math.ceil(math.sqrt(num_frames))
       rows = math.ceil(num_frames / cols)
       
       sheet_w = (sprite_w * cols) + (padding * (cols + 1))
       sheet_h = (sprite_h * rows) + (padding * (rows + 1))
       
       self.sprite_sheet_img = Image.new("P", (sheet_w, sheet_h), color=15)
       self.sprite_sheet_img.putpalette(self.master_palette)
       
       for index, frame in enumerate(self.processed_frames):
           c_col = index % cols
           c_row = index // cols
           x = (c_col * sprite_w) + (padding * (c_col + 1))
           y = (c_row * sprite_h) + (padding * (c_row + 1))
           self.sprite_sheet_img.paste(frame, (x, y))
           
       # Refresh screen renders
       self.render_atlas_viewport()
       self.setup_animation_loop()
       
       # Enable interface operations
       self.btn_compile.configure(state="normal")
       self.btn_export_gif.configure(state="normal")
       self.btn_export_html.configure(state="normal")
       
       # Garbage collect and flush unused objects to keep memory usage low on 8GB machines
       gc.collect()
       
   def apply_bayer_dither(self, img_pil, pal_template):
       """Applies a classic 4x4 Bayer ordered dither matrix to an RGB image."""
       width, height = img_pil.size
       pixels = img_pil.load()
       
       # Standard Bayer 4x4 matrix normalized values (scale: 0-255)
       bayer_matrix = [
           [  0, 128,  32, 160],
           [192,  64, 224,  96],
           [ 48, 176,  16, 144],
           [240, 112, 208,  80]
       ]
       
       # Apply matrix offset directly to channel colors to create retro halftone meshes
       dithered_img = Image.new("RGB", (width, height))
       dithered_pix = dithered_img.load()
       
       for y in range(height):
           for x in range(width):
               factor = (bayer_matrix[y % 4][x % 4] / 255.0) - 0.5
               r, g, b = pixels[x, y]
               
               # Apply dither shift factor
               offset = int(factor * 48)  # Strength constant
               dithered_pix[x, y] = (
                   max(0, min(255, r + offset)),
                   max(0, min(255, g + offset)),
                   max(0, min(255, b + offset))
               )
               
       # Remap image onto the palette boundary
       return dithered_img.quantize(palette=pal_template, dither=Image.Dither.NONE)
       
   def render_atlas_viewport(self):
       if not self.sprite_sheet_img:
           return
           
       self.atlas_canvas.delete("all")
       canvas_w = self.atlas_canvas.winfo_width()
       canvas_h = self.atlas_canvas.winfo_height()
       
       if canvas_w < 10 or canvas_h < 10:
           self.root.after(100, self.render_atlas_viewport)
           return
           
       # Convert indexed file output into viewable RGBA structure
       preview_rgba = self.make_rgba_preview(self.sprite_sheet_img)
       img_w, img_h = preview_rgba.size
       
       scale = min(canvas_w / img_w, canvas_h / img_h) * 0.95
       scale = max(1.0, scale)
       
       target_w = int(img_w * scale)
       target_h = int(img_h * scale)
       
       scaled_preview = preview_rgba.resize((target_w, target_h), Image.Resampling.NEAREST)
       self.atlas_photo = ImageTk.PhotoImage(scaled_preview)
       
       # Render background transparent checker grid
       self.draw_checkerboard(self.atlas_canvas, canvas_w, canvas_h)
       
       # Draw on screen
       self.atlas_canvas.create_image(canvas_w // 2, canvas_h // 2, image=self.atlas_photo, anchor="center")
       
   def setup_animation_loop(self):
       self.preview_tk_images = []
       for img in self.processed_frames:
           rgba_img = self.make_rgba_preview(img)
           self.preview_tk_images.append(rgba_img)
           
       # Align index markers
       if self.animation_index >= len(self.preview_tk_images):
           self.animation_index = 0
           
       self.loop_animation()
       
   def loop_animation(self):
       if not self.preview_tk_images:
           return
           
       if self.animation_job:
           self.root.after_cancel(self.animation_job)
           self.animation_job = None
           
       self.preview_canvas.delete("all")
       canvas_w = self.preview_canvas.winfo_width()
       canvas_h = self.preview_canvas.winfo_height()
       
       if canvas_w < 10 or canvas_h < 10:
           self.animation_job = self.root.after(100, self.loop_animation)
           return
           
       self.draw_checkerboard(self.preview_canvas, canvas_w, canvas_h)
       
       current_frame = self.preview_tk_images[self.animation_index]
       img_w, img_h = current_frame.size
       
       # Calculate viewport scale limits
       scale = min(canvas_w / img_w, canvas_h / img_h) * 0.75
       scale = max(1.0, scale)
       
       scaled_w = int(img_w * scale)
       scaled_h = int(img_h * scale)
       
       resized_frame = current_frame.resize((scaled_w, scaled_h), Image.Resampling.NEAREST)
       self.preview_photo = ImageTk.PhotoImage(resized_frame)
       
       cx, cy = canvas_w // 2, canvas_h // 2
       self.preview_canvas.create_image(cx, cy, image=self.preview_photo, anchor="center")
       
       # Draw registration alignment crosshair over preview
       self.preview_canvas.create_line(cx - 15, cy, cx + 15, cy, fill="#FF453A", width=1)
       self.preview_canvas.create_line(cx, cy - 15, cx, cy + 15, fill="#FF453A", width=1)
       
       # Update timeline text reads
       dx, dy = self.offsets[self.animation_index]
       self.frame_info_var.set(f"Selected: Frame {self.animation_index + 1}/{len(self.preview_tk_images)} | Pivot Offset Adjust -> X: {dx}px, Y: {dy}px (Drag to adjust)")
       
       # Schedule next tick cycle
       self.animation_index = (self.animation_index + 1) % len(self.preview_tk_images)
       fps = self.fps_var.get()
       delay_ms = int(1000 / fps)
       self.animation_job = self.root.after(delay_ms, self.loop_animation)
       
   def start_registration_drag(self, event):
       """Initializes dragging to offset pivot coordinates."""
       if not self.preview_tk_images:
           return
       # Freeze frame index to edit currently active sprite slice
       self.selected_timeline_index = (self.animation_index - 1) % len(self.preview_tk_images)
       self.drag_start_x = event.x
       self.drag_start_y = event.y
       
   def execute_registration_drag(self, event):
       """Calculates displacement factors and updates frame alignment offsets."""
       if not self.preview_tk_images:
           return
           
       dx = event.x - self.drag_start_x
       dy = event.y - self.drag_start_y
       
       # Set alignment threshold
       if abs(dx) > 3 or abs(dy) > 3:
           curr_dx, curr_dy = self.offsets[self.selected_timeline_index]
           # Accumulate drag offsets to shift crop alignment bounds
           self.offsets[self.selected_timeline_index] = (curr_dx + int(dx // 4), curr_dy + int(dy // 4))
           
           self.drag_start_x = event.x
           self.drag_start_y = event.y
           
           # Instantly recompute and redraw frames
           self.recompute()
           
   def make_rgba_preview(self, pil_img):
       """Creates transparent RGBA views for standard displays by mapping #FF00FF to transparent."""
       rgba = pil_img.convert("RGBA")
       raw_pixels = rgba.getdata()
       cleaned_data = []
       for pixel in raw_pixels:
           if pixel[0] == 255 and pixel[1] == 0 and pixel[2] == 255:
               cleaned_data.append((0, 0, 0, 0)) # Fully transparent pixel
           else:
               cleaned_data.append(pixel)
       rgba.putdata(cleaned_data)
       return rgba
       
   def draw_checkerboard(self, canvas, width, height, tile_size=14):
       """Draws a professional checkerboard grid beneath sprite elements."""
       color1 = "#1A1A1A"
       color2 = "#141414"
       for y in range(0, height, tile_size):
           for x in range(0, width, tile_size):
               fill_color = color1 if ((x // tile_size) + (y // tile_size)) % 2 == 0 else color2
               canvas.create_rectangle(x, y, x + tile_size, y + tile_size, fill=fill_color, outline="")

6. Edge Case Fail-Safe Procedures
To prevent application errors, verify the following parameters during complex compile procedures:
Detected Issue
	Root Cause
	Resolution Strategy
	Bilinear Blur on Safari Output
	Modern browser defaults blur scaled canvas elements.
	The template has explicit image-rendering: pixelated; CSS overrides and sets ctx.imageSmoothingEnabled = false globally inside the script. Do not modify these rules.
	System Out of Memory (OOM)
	High-resolution frames saturate M1 unified memory.
	The processing engine is written sequentially, discarding original image pointers after scaling. Avoid loading multi-gigabyte folder structures.
	Flickering Colors Between Frames
	Distinct palettes compiled per individual frames.
	Resolved via global consolidation step (composite compilation) before running quantization. This forces all animation frames to share the same color indices.
